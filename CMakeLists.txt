cmake_minimum_required(VERSION 3.22)
project(scrctl_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(XDB_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(XDB_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(XDB_INSTALL OFF CACHE BOOL "" FORCE)
set(XDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(XDB_FETCH_DEPENDENCIES ON CACHE BOOL "" FORCE)
set(XDB_LIBAV ON CACHE BOOL "" FORCE)
set(XDB_LIBAV_AUDIO OFF CACHE BOOL "" FORCE)
set(XDB_LIBAVFORMAT OFF CACHE BOOL "" FORCE)
add_subdirectory(xdb)
# App-specific libcurl defaults without changing any pinned SDK source.
target_sources(xdb_server PRIVATE app/src/main/cpp/CurlTrust.cpp)
target_link_options(xdb_server PRIVATE "LINKER:--wrap=curl_easy_init")
add_library(scrctl_jni SHARED app/src/main/cpp/Bridge.cpp)
target_link_libraries(scrctl_jni PRIVATE xdb_client xdb_server nlohmann_json::nlohmann_json jnigraphics log)
target_compile_options(scrctl_jni PRIVATE -Wall -Wextra -Werror)
